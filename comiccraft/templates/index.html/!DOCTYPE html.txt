    <!-- Client-side JavaScript Orchestration Layer -->
    <script>
        async function generateComic() {
            const promptInput = document.getElementById('prompt').value.trim();
            const panelSelect = document.getElementById('panels').value;
            const generateBtn = document.getElementById('generateBtn');
            const statusBlock = document.getElementById('statusBlock');

            if (!promptInput) {
                alert('Please enter a story premise first!');
                return;
            }

            // Lock UI controls down during generation lifecycle
            generateBtn.disabled = true;
            generateBtn.classList.add('opacity-50', 'cursor-not-allowed');
            statusBlock.classList.remove('hidden');

            try {
                const response = await fetch('/generate-story', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ 
                        prompt: promptInput,
                        panel_count: parseInt(panelSelect)
                    })
                });

                if (!response.ok) {
                    throw new Error('Failed to generate script layout.');
                }

                const data = await response.json();
                console.log('Story Script Generated:', data);
                
                // Transition user over to the visual canvas rendering layout dashboard 
                window.location.href = '/preview';

            } catch (error) {
                alert('Error crafting script: ' + error.message);
                generateBtn.disabled = false;
                generateBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                statusBlock.classList.add('hidden');
            }
        }
    </script>
