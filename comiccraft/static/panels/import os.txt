import os
from PIL import Image, ImageDraw

def create_mock_panel(output_dir="static/panels", filename="panel_1.png"):
    """
    Generates a stylized placeholder panel to test the front-end grid presentation.
    """
    # Ensure target structural cache directory path exists
    os.makedirs(output_dir, exist_ok=True)
    full_path = os.path.join(output_dir, filename)

    # 1. Initialize a 512x512 high-contrast dark square canvas
    width, height = 512, 512
    image = Image.new("RGB", (width, height), color="#1e1e24")
    draw = ImageDraw.Draw(image)

    # 2. Draw standard inner framing borders
    border_margin = 16
    draw.rectangle(
        [border_margin, border_margin, width - border_margin, height - border_margin],
        outline="#374151",
        width=3
    )

    # 3. Add an abstract geometric focal point (simulating composition layouts)
    center_x, center_y = width // 2, height // 2
    radius = 80
    draw.ellipse(
        [center_x - radius, center_y - radius, center_x + radius, center_y + radius],
        fill="#4f46e5"
    )

    # Save asset matrix onto physical disc mapping
    image.save(full_path, "PNG")
    print(f"Successfully generated placeholder asset at: {full_path}")

if _name_ == "_main_":
    # Adjust path references relative to your repository root executing terminal context
    create_mock_panel()
