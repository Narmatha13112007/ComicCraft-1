import os
import time
from flask import current_app

def cleanup_expired_exports(max_age_seconds=7200):
    """
    Scans the static export file matrix and purges compiled PDF or ZIP bundles
    older than the specified expiration duration to conserve server storage space.
    """
    # Safeguard retrieval from application configurations setup in main.py
    exports_dir = current_app.config.get('EXPORT_FOLDER', 'static/exports')
    
    if not os.path.exists(exports_dir):
        return
        
    current_time = time.time()
    purged_count = 0

    try:
        for filename in os.listdir(exports_dir):
            file_path = os.path.join(exports_dir, filename)
            
            # Skip hidden tracking files like .gitkeep
            if filename.startswith('.'):
                continue
                
            # Inspect system modification metrics 
            file_age = current_time - os.path.getmtime(file_path)
            
            if file_age > max_age_seconds:
                os.remove(file_path)
                purged_count += 1
                
        if purged_count > 0:
            current_app.logger.info(f"Storage clean cycle complete. Purged {purged_count} expired export files.")
            
    except Exception as e:
        print(f"Error executing storage sweep: {e}")
